handleSaveAppointment = async (formData) => {
  // 1. Valida RBAC
  if (currentRole === 'recepcao' && formData.value) {
    throw new Error('Sem permissão para editar valor');
  }

  // 2. Prepara dados
  const appointmentData = {
    clinic_id: clinicId,
    patient_id: formData.patient_id,
    ...
  };

  // 3. Cria ou edita
  if (agenda.selectedSlot?.id) {
    const result = await updateAppointment(...);
  } else {
    const result = await createAppointment(...);
  }

  // 4. Atualiza estado local
  agenda.addAppointmentLocal(result);

  // 5. Recarrega lista
  await loadAgendaData();

  // 6. Fecha modal
  agenda.deselectSlot();
};
