2. USAR HOOK USECHECK-IN DIRETO

```tsx
import { useCheckIn } from '@/modules/agenda/reception';

export function CustomCheckIn() {
  const { loading, error, success, perform_checkin } = useCheckIn(clinic_id);
  
  async function handleClick() {
    await perform_checkin({
      appointment_id: '123',
      notes: 'Chegou puntualmente'
    });
  }
  
  return (
    <button onClick={handleClick} disabled={loading}>
      {success ? '✅ Realizado' : 'Check-in'}
    </button>
  );
}
```

3. USAR API DIRETAMENTE

```tsx
import { performCheckIn, getWaitingQueue } from '@/modules/agenda/reception';

async function myFunction() {
  // Fazer check-in
  const result = await performCheckIn({
    appointment_id: '123',
    clinic_id: 'clinic-1',
    checked_in_by: user.id,
    notes: 'Opcional'
  });
  
  if (result.success) {
    console.log('Check-in:', result.checkin_id);
  }
  
  // Buscar fila
  const { queue } = await getWaitingQueue({
    clinic_id: 'clinic-1',
    limit: 50
  });
}
```
