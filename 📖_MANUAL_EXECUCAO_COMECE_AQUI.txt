💡 COMO USAR OS COMPONENTES

// Import
import {
  OfficialStatusBadge,
  OfficialStatusSelect,
  OperationalTimeline
} from '@/modules/agenda';

// Usar em JSX

// 1. Mostrar status
<OfficialStatusBadge status="completed" size="md" />
// Resultado: [✔️ Completo]

// 2. Mudar status
<OfficialStatusSelect
  currentStatus={appointment.official_status}
  onStatusChange={handleStatusChange}
/>
// Valida automaticamente, mostra só transições válidas

// 3. Mostrar progresso
<OperationalTimeline
  currentStatus="in_progress"
  completionHistory={statusHistory}
/>
// Timeline visual do progresso

// 4. Validar antes de salvar
import { validateStatusTransition } from '@/modules/agenda';

const result = validateStatusTransition('scheduled', 'confirmed');
if (result.valid) {
  // Pode mudar
}

// 5. Usar filtros rápidos
import { QUICK_FILTERS, getQuickFilterSummary } from '@/modules/agenda';

const pending = appointments.filter(
  apt => QUICK_FILTERS.PENDING.includes(apt.official_status)
);

const summary = getQuickFilterSummary(appointments);
console.log(summary); // { pending: 10, finalized: 5, ... }
