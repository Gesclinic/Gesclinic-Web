```tsx
import React from 'react';
import { useClinicContext } from '@/contexts/ClinicContext';
import { OperationalDashboard, WaitingQueuePanel } from '@/modules/agenda/reception';

export function ReceptionPage() {
  const { clinicId } = useClinicContext();

  if (!clinicId) {

  return (
    <div className="min-h-screen bg-gray-50 p-6">
      {/* Header */}
      <div className="mb-8">
        <h1 className="text-3xl font-bold text-gray-900">📋 Recepção</h1>
        <p className="text-gray-600">Gerenciamento de check-in e fila de espera</p>
      </div>

      {/* Main Dashboard */}
      <div className="grid grid-cols-1 gap-8 lg:grid-cols-3">
        {/* Dashboard Principal (2/3) */}
        <div className="lg:col-span-2">
          <OperationalDashboard clinic_id={clinicId} />
        </div>

        {/* Fila Rápida (1/3) */}
        <div>
          <WaitingQueuePanel
            clinic_id={clinicId}
            maxItems={10}
            showSearch={false}
          />
        </div>
      </div>
    </div>
  );
}

export default ReceptionPage;
```

```tsx
// components/ReceptionStats.tsx

import React from 'react';
import { useWaitingQueue } from '@/modules/agenda/reception';

interface ReceptionStatsProps {
  clinic_id: string;
}

export function ReceptionStats({ clinic_id }: ReceptionStatsProps) {
  const { queue, stats, is_live, loading } = useWaitingQueue(clinic_id);

  if (loading) {
    return <div className="animate-pulse">Carregando...</div>;
  }

  return (
    <div className="grid grid-cols-4 gap-4">
      {/* Total */}
      <div className="rounded-lg bg-white p-6 shadow-sm border border-gray-100">
        <div className="text-2xl font-bold text-purple-600">{queue.length}</div>
        <p className="text-sm text-gray-600 mt-1">Pacientes na fila</p>
      </div>

      {/* Tempo Médio */}
      <div className="rounded-lg bg-white p-6 shadow-sm border border-gray-100">
        <div className="text-2xl font-bold text-blue-600">
          {stats.average_wait_time_minutes}
        </div>
        <p className="text-sm text-gray-600 mt-1">Tempo médio (min)</p>
      </div>

      {/* Tempo Máximo */}
      <div className="rounded-lg bg-white p-6 shadow-sm border border-gray-100">
        <div className="text-2xl font-bold text-orange-600">
          {stats.max_wait_time_minutes}
        </div>
        <p className="text-sm text-gray-600 mt-1">Tempo máximo (min)</p>
      </div>

      {/* Críticos */}
      <div className="rounded-lg bg-white p-6 shadow-sm border border-gray-100">
        <div className={`text-2xl font-bold ${
          stats.critical_count > 0 ? 'text-red-600' : 'text-green-600'
        }`}>
          {stats.critical_count}
        </div>
        <p className="text-sm text-gray-600 mt-1">
          {stats.critical_count > 0 ? 'Em crítico' : 'Nenhum crítico'}
        </p>
        {is_live && <span className="text-xs text-green-600 mt-2">● AO VIVO</span>}
      </div>
    </div>
  );
}
```

```tsx
// components/SyncStatus.tsx

import React, { useEffect } from 'react';
import { useReceptionRealtimeSync } from '@/modules/agenda/reception';

interface SyncStatusProps {
  clinic_id: string;
}

export function SyncStatus({ clinic_id }: SyncStatusProps) {
  const { sync_state, on_event, off_event } = useReceptionRealtimeSync(clinic_id);

  useEffect(() => {
    const handler = (event: any) => {
      console.log('Evento recebido:', event);
    };

    on_event(handler);
    return () => off_event(handler);
  }, [on_event, off_event]);

  return (
    <div className="flex items-center gap-3 text-sm">
      {/* Status */}
      <div className="flex items-center gap-2">
        <div
          className={`h-2 w-2 rounded-full ${
            sync_state.is_connected ? 'bg-green-500' : 'bg-red-500'
          }`}
        />
        <span className="text-gray-600">
          {sync_state.is_connected ? 'Conectado' : 'Desconectado'}
        </span>
      </div>

      {/* Pendentes */}
      {sync_state.pending_updates > 0 && (
        <div className="flex items-center gap-2 text-yellow-600">
          <span>⏳ {sync_state.pending_updates} pendentes</span>
        </div>
      )}

      {/* Erro */}
      {sync_state.error && (
        <div className="flex items-center gap-2 text-red-600">
          <span>⚠️ {sync_state.error}</span>
        </div>
      )}

      {/* Último sync */}
      <span className="text-xs text-gray-500">
        Última atualização: {new Date(sync_state.last_sync).toLocaleTimeString('pt-BR')}
      </span>
    </div>
  );
}
```

```tsx
// hooks/useReceptionNotifications.ts

import { useEffect } from 'react';
import { useWaitingQueue } from '@/modules/agenda/reception';

export function useReceptionNotifications(clinic_id: string) {
  const { queue, stats } = useWaitingQueue(clinic_id);

  useEffect(() => {
    // Notificar quando crítico aumenta
    if (stats.critical_count > 0) {
      const message = `⚠️ ${stats.critical_count} paciente(s) aguardando > 30min`;
      
      // Usar Notification API
      if ('Notification' in window && Notification.permission === 'granted') {
        new Notification('Recepção - Alerta', {
          body: message,
          icon: '/favicon.ico',
        });
      }
    }
  }, [stats.critical_count]);

  return {
    has_critical: stats.critical_count > 0,
    critical_count: stats.critical_count,
    warning_count: stats.warning_count,
  };
}
```

```tsx
// pages/Agenda/AgendaPage.tsx

import { useMemo } from 'react';
import { CheckInButton } from '@/modules/agenda/reception';
import { useAppointments } from '@/hooks/useAppointments';
import { useClinicContext } from '@/contexts/ClinicContext';

export function AgendaPage() {
  const { clinicId } = useClinicContext();
  const { appointments } = useAppointments(clinicId);

  // Filtrar apenas "confirmed" para mostrar CheckIn
  const confirmedAppointments = useMemo(
    () => appointments.filter(apt => apt.official_status === 'confirmed'),
    [appointments]
  );

  return (
    <div className="space-y-4">
      {confirmedAppointments.map(apt => (
        <div key={apt.id} className="flex items-center justify-between p-4 border rounded">
          <div>
            <h4 className="font-semibold">{apt.patient_name}</h4>
            <p className="text-sm text-gray-600">
              {new Date(apt.scheduled_date).toLocaleString('pt-BR')}
            </p>
          </div>

          <CheckInButton
            appointment_id={apt.id}
            clinic_id={clinicId}
            status={apt.official_status}
            patient_name={apt.patient_name}
            onSuccess={() => {
              // Refetch appointments
            }}
          />
        </div>
      ))}
    </div>
  );
}
```
