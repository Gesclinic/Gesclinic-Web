1. Páginas:
   ```jsx
   export const ServicosPage = () => {
     const { clinicId, userRole } = useAuth();
     const { data, loading, error } = useServices(clinicId);
     
     if (!canAccess(userRole, 'base:servicos')) {
       return <AccessDenied />;
     }
     
     return (
       <BaseDataTable
         title="Serviços/Procedimentos"
         data={data}
         columns={servicosColumns}
         onNew={() => navigate('novo')}
       />
     );
   };
   ```

2. Queries (em clinicsApi.js ou novo baseApi.js):
   ```javascript
   export const getServices = async (clinicId) => {
     const { data, error } = await supabase
       .from('services')
       .select('id, name, description, category')  // ✓ Apenas colunas existentes
       .eq('clinic_id', clinicId)
       .eq('deleted_at', null);  // Soft delete
     
     if (error) throw error;
     return data;
   };
   ```
