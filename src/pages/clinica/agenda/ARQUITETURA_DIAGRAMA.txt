GERENCIAMENTO DE ESTADO:
════════════════════════

const agenda = useAgendaStore()

LEITURA:
  agenda.date              // String (YYYY-MM-DD)
  agenda.viewMode          // 'geral' | 'profissional' | 'sala'
  agenda.filters           // { professional, room, status, payer, service, searchText }
  agenda.appointments      // Array completo
  agenda.filteredAppointments  // Array filtrado
  agenda.selectedSlot      // Agendamento selecionado
  agenda.indicators        // { total, confirmed, noShow, fitting, occupationRate }
  agenda.metadata          // { professionals, rooms, services, payers, patients }

ESCRITA:
  agenda.setDate(date)
  agenda.setViewMode(mode)
  agenda.updateFilter(key, value)
  agenda.setMultipleFilters(obj)
  agenda.clearFilters()
  agenda.selectSlot(slot)
  agenda.deselectSlot()
  agenda.setAppointments(arr)
  agenda.updateAppointmentLocal(id, updates)
  agenda.removeAppointmentLocal(id)
  agenda.addAppointmentLocal(apt)
