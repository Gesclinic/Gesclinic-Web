FLUXO OPERACIONAL DE RECEPÇÃO:

  1. AGENDADO         → Agendamento criado (status: scheduled)
  2. CONFIRMADO       → Paciente confirmou (status: confirmed)
  3. ✅ CHECK-IN      → Paciente chega na recepção (status: checked_in) ⭐ NOVO
  4. ⏳ AGUARDANDO     → Fila de espera até profissional disponível (status: waiting)
  5. 🔄 ATENDIMENTO   → Sendo atendido (status: in_progress)
  6. ✔️ FINALIZADO    → Atendimento concluído (status: completed)

INTEGRAÇÃO COM STATUS OFICIAL MODEL:
  ✅ Usa status oficial existentes (não cria novos)
  ✅ Apenas organiza o fluxo operacional
  ✅ Adiciona data/hora/usuário ao check-in
  ✅ Realtime para atualização automática

SQL A EXECUTAR:

-- Tabela de check-ins
CREATE TABLE reception_checkins (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  appointment_id UUID NOT NULL REFERENCES appointments(id) ON DELETE CASCADE,
  clinic_id UUID NOT NULL REFERENCES clinics(id) ON DELETE CASCADE,
  checked_in_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  checked_in_by UUID NOT NULL,
  created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT NOW(),
  UNIQUE(appointment_id) -- Um check-in por agendamento
);

CREATE INDEX idx_reception_checkins_clinic_id 
ON reception_checkins(clinic_id);

CREATE INDEX idx_reception_checkins_appointment_id 
ON reception_checkins(appointment_id);

CREATE INDEX idx_reception_checkins_checked_in_at 
ON reception_checkins(clinic_id, checked_in_at DESC);

-- View de fila de espera
CREATE OR REPLACE VIEW waiting_queue_view AS
SELECT 
  a.id,
  a.appointment_id_original,
  a.clinic_id,
  a.patient_id,
  a.scheduled_date,
  a.official_status,
  rc.checked_in_at,
  rc.checked_in_by,
  EXTRACT(EPOCH FROM (NOW() - rc.checked_in_at))::INTEGER as tempo_espera_segundos,
  FLOOR(EXTRACT(EPOCH FROM (NOW() - rc.checked_in_at)) / 60)::INTEGER as tempo_espera_minutos
FROM appointments a
LEFT JOIN reception_checkins rc ON a.id = rc.appointment_id
WHERE a.clinic_id = a.clinic_id
  AND a.official_status IN ('checked_in', 'waiting')
ORDER BY rc.checked_in_at ASC NULLS LAST;

-- RLS Policies
ALTER TABLE reception_checkins ENABLE ROW LEVEL SECURITY;

CREATE POLICY "usuarios_podem_ver_checkins_sua_clinica" ON reception_checkins
  FOR SELECT USING (
    clinic_id IN (
      SELECT clinic_id FROM user_clinic_roles 
      WHERE user_id = auth.uid()
    )
  );

CREATE POLICY "usuarios_podem_criar_checkins" ON reception_checkins
  FOR INSERT WITH CHECK (
    clinic_id IN (
      SELECT clinic_id FROM user_clinic_roles 
      WHERE user_id = auth.uid()
    )
  );

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. useCheckIn()
   ├─ Função: performCheckIn(appointmentId)
   ├─ Valida: status = 'confirmed'
   ├─ POST: /reception/checkin
   ├─ Otimista: atualiza UI imediato
   ├─ Invalidates: waitingQueue query
   └─ Retorna: { loading, error, success }

2. useWaitingQueue()
   ├─ Query: SELECT * FROM waiting_queue_view
   ├─ Realtime: subscriptions.on('*')
   ├─ Auto-refresh: a cada 5 segundos
   ├─ Cálculo: tempo_espera_minutos
   ├─ Filtro: clinic_id
   └─ Retorna: { queue, loading, isLive }

3. useReceptionOperations()
   ├─ Query: múltiplas (próximos, atendimento, finalizados)
   ├─ Realtime: todas as seções
   ├─ Agregação: totais por status
   ├─ Cache: 30 segundos
   └─ Retorna: { operationalData, loading }

4. useReceptionRealtimeSync()
   ├─ Monitora: receipt_checkins table
   ├─ Subscriptions: INSERT, UPDATE, DELETE
   ├─ Broadcast: envia eventos para outras abas
   ├─ Deduplica: mesma mudança de múltiplas fontes
   └─ Retorna: { syncStatus, lastUpdate }
