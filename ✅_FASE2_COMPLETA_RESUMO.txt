EXEMPLO 3: Usar Hook Diretamente

```tsx
import { useWaitingQueue } from '@/modules/agenda/reception';

export function CustomQueue() {
  const { queue, stats, loading, is_live } = useWaitingQueue('clinic-123');
  
  if (loading) return <div>Carregando...</div>;
  
  return (
    <div>
      <p>Na fila: {queue.length}</p>
      <p>Tempo médio: {stats.average_wait_time_minutes}min</p>
      <p>Status: {is_live ? '🟢 AO VIVO' : '🔴 Offline'}</p>
    </div>
  );
}
```


EXEMPLO 4: Sincronização Entre Abas
─────────────────────────────────────

```tsx
import { useReceptionRealtimeSync } from '@/modules/agenda/reception';

export function SyncMonitor() {
  const { sync_state, on_event, off_event } = useReceptionRealtimeSync('clinic-123');
  
  useEffect(() => {
    const handler = (event) => {
      console.log('Evento sincronizado:', event);
    };
    
    on_event(handler);
    return () => off_event(handler);
  }, [on_event, off_event]);
  
  return (
    <div>
      <p>Conectado: {sync_state.is_connected ? '✅' : '❌'}</p>
      <p>Pendente: {sync_state.pending_updates}</p>
      {sync_state.error && <p className="text-red-600">{sync_state.error}</p>}
    </div>
  );
}
```
